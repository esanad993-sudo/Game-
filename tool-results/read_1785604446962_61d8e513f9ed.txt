    96→// ─── API helpers ───
    97→const api = {
    98→  get: async (url: string) => {
    99→    const res = await fetch(url)
   100→    if (!res.ok) throw new Error(`GET ${url} failed`)
   101→    return res.json()
   102→  },
   103→  post: async (url: string, data: any) => {
   104→    const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   105→    if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || `POST ${url} failed`) }
   106→    return res.json()
   107→  },
   108→  patch: async (url: string, data: any) => {
   109→    const res = await fetch(url, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   110→    if (!res.ok) throw new Error(`PATCH ${url} failed`)
   111→    return res.json()
   112→  },
   113→  del: async (url: string) => {
   114→    const res = await fetch(url, { method: 'DELETE' })
   115→    if (!res.ok) throw new Error(`DELETE ${url} failed`)
   116→    return res.json()
   117→  },
   118→}

   142→    // Read from localStorage on mount - this is intentional initialization
   143→    const saved = localStorage.getItem('gameforge_user') || localStorage.getItem('turborush_user')
   144→    if (saved) {
   145→      try {
   146→        const parsed = JSON.parse(saved)
   147→        // Use a microtask to avoid synchronous setState in effect
   148→        queueMicrotask(() => setUser(normalizeUser(parsed)))
   149→      } catch {}
   150→    }
   151→    queueMicrotask(() => setMounted(true))
   152→  }, [])
   153→
   154→  const login = (u: User) => {
   155→    const safe = normalizeUser(u)
   156→    setUser(safe)

   159→
   160→  const logout = () => {
   161→    setUser(null)
   162→    localStorage.removeItem('gameforge_user')
   163→    localStorage.removeItem('turborush_user')

   165→  }

   167→  if (!mounted) {
   168→    return (
   169→      <div className="min-h-screen flex items-center justify-center gf-bg">

   904→// ─── Game Player (simplified for now — will port the full WebGL game) ───
   905→function GamePlayer({ questions, mode, playerName, userId, setId, onEnd }: {
   906→  questions: Question[]
   907→  mode: GameModeId
   908→  playerName: string
   909→  userId: string
   910→  setId: string
   911→  onEnd: () => void
   912→}) {
   913→  const [currentQ, setCurrentQ] = useState(0)
   914→  const [score, setScore] = useState(0)
   915→  const [correct, setCorrect] = useState(0)
   916→  const [wrong, setWrong] = useState(0)
   917→  const [streak, setStreak] = useState(0)
   918→  const [bestStreak, setBestStreak] = useState(0)
   919→  const [hearts, setHearts] = useState(3)
   920→  const [coins, setCoins] = useState(0)
   921→  const [showResult, setShowResult] = useState(false)
   922→  const [lastAnswer, setLastAnswer] = useState<'correct' | 'wrong' | null>(null)
   923→  const [gameOver, setGameOver] = useState(false)
   924→  const [timeLeft, setTimeLeft] = useState(15)
   925→  const [answered, setAnswered] = useState(false)
   926→
   927→  // Timer
   928→  useEffect(() => {
   929→    if (gameOver || answered) return
   930→    const timer = setInterval(() => {
   931→      setTimeLeft(prev => {
   932→        if (prev <= 1) {
   933→          handleAnswer(-1) // timeout
   934→          return 0
   935→        }
   936→        return prev - 1
   937→      })
   938→    }, 1000)
   939→    return () => clearInterval(timer)
   940→  }, [currentQ, gameOver, answered])
   941→
   942→  const handleAnswer = (idx: number) => {
   943→    if (answered) return
   944→    setAnswered(true)
   945→
   946→    const q = questions[currentQ]
   947→    const isCorrect = idx === q.correctIdx
   948→
   949→    if (isCorrect) {
   950→      const newStreak = streak + 1
   951→      const points = 50 + newStreak * 25
   952→      setScore(s => s + points)
   953→      setCorrect(c => c + 1)
   954→      setStreak(newStreak)
   955→      setBestStreak(bs => Math.max(bs, newStreak))
   956→      setCoins(c => c + 5)
   957→      setLastAnswer('correct')
   958→    } else {
   959→      setWrong(w => w + 1)
   960→      setStreak(0)
   961→      if (mode === 'battle-royale' || mode === 'classic') {
   962→        setHearts(h => {
   963→          const newH = h - 1
   964→          if (newH <= 0) setGameOver(true)
   965→          return newH
   966→        })
   967→      }
   968→      setLastAnswer('wrong')
   969→    }
   970→
   971→    setShowResult(true)
   972→    setTimeout(() => {
   973→      setShowResult(false)
   974→      setLastAnswer(null)
   975→      setAnswered(false)
   976→      setTimeLeft(15)
   977→      if (currentQ + 1 >= questions.length) {
   978→        setGameOver(true)
   979→      } else {
   980→        setCurrentQ(q => q + 1)
   981→      }
   982→    }, 1500)
   983→  }
   984→
   985→  // Save result on game over
   986→  useEffect(() => {
   987→    if (gameOver) {
   988→      api.post('/api/game-results', {
   989→        userId,
   990→        gameType: 'solo',
   991→        setId,
   992→        score,
   993→        correct,
   994→        wrong,
   995→        bestStreak,
   996→        coinsEarned: coins,
   997→      }).catch(() => {})
   998→    }
   999→  }, [gameOver])
  1000→
  1001→  const modeConfig = getGameMode(mode)
  1002→
  1003→  if (gameOver) {
  1004→    return (
  1005→      <div className="gf-card p-8 text-center max-w-lg mx-auto">
  1006→        <motion.div initial={{ scale: 0.5 }} animate={{ scale: 1 }} className="text-6xl mb-4">🏁</motion.div>
  1007→        <h2 className="gf-font-display text-2xl text-gf-dark mb-4">Game Over!</h2>
  1008→        <div className="grid grid-cols-2 gap-4 mb-6">
  1009→          <div className="p-3 rounded-xl bg-gf-light/50">
  1010→            <p className="gf-font-display text-2xl text-gf-dark">{score.toLocaleString()}</p>
  1011→            <p className="text-xs text-gf-dark/60 font-bold">Score</p>
  1012→          </div>
  1013→          <div className="p-3 rounded-xl bg-gf-success/20">
  1014→            <p className="gf-font-display text-2xl text-gf-dark">{correct}/{correct + wrong}</p>
  1015→            <p className="text-xs text-gf-dark/60 font-bold">Correct</p>
  1016→          </div>
  1017→          <div className="p-3 rounded-xl bg-gf-warning/20">
  1018→            <p className="gf-font-display text-2xl text-gf-dark">{bestStreak}</p>
  1019→            <p className="text-xs text-gf-dark/60 font-bold">Best Streak</p>
  1020→          </div>
  1021→          <div className="p-3 rounded-xl bg-gf-purple/20">
  1022→            <p className="gf-font-display text-2xl text-gf-dark">🪙 {coins}</p>
  1023→            <p className="text-xs text-gf-dark/60 font-bold">Coins Earned</p>
  1024→          </div>
  1025→        </div>
  1026→        <div className="flex gap-3 justify-center">
  1027→          <button onClick={onEnd} className="gf-btn text-sm">🏠 Dashboard</button>
  1028→          <button onClick={() => {
  1029→            setCurrentQ(0); setScore(0); setCorrect(0); setWrong(0); setStreak(0); setBestStreak(0); setHearts(3); setCoins(0); setGameOver(false); setAnswered(false); setTimeLeft(15)
