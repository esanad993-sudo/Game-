    94→// ─── API helpers ───
    95→const api = {
    96→  get: async (url: string) => {
    97→    const res = await fetch(url)
    98→    if (!res.ok) throw new Error(`GET ${url} failed`)
    99→    return res.json()
   100→  },
   101→  post: async (url: string, data: any) => {
   102→    const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   103→    if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || `POST ${url} failed`) }
   104→    return res.json()
   105→  },
   106→  patch: async (url: string, data: any) => {
   107→    const res = await fetch(url, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   108→    if (!res.ok) throw new Error(`PATCH ${url} failed`)
   109→    return res.json()
   110→  },
   111→  del: async (url: string) => {
   112→    const res = await fetch(url, { method: 'DELETE' })
   113→    if (!res.ok) throw new Error(`DELETE ${url} failed`)
   114→    return res.json()
   115→  },
   116→}

   119→export default function Home() {
   120→  const [user, setUser] = useState<User | null>(null)
   121→  const [view, setView] = useState<View>('home')
   122→  const [mounted, setMounted] = useState(false)
   123→
   124→  useEffect(() => {
   125→    // Read from localStorage on mount - this is intentional initialization
   126→    const saved = localStorage.getItem('turborush_user')
   127→    if (saved) {
   128→      try {
   129→        const parsed = JSON.parse(saved)
   130→        // Use a microtask to avoid synchronous setState in effect
   131→        queueMicrotask(() => setUser(parsed))
   132→      } catch {}
   133→    }
   134→    queueMicrotask(() => setMounted(true))
   135→  }, [])
   136→
   137→  const login = (u: User) => {
   138→    setUser(u)
   139→    localStorage.setItem('turborush_user', JSON.stringify(u))
   140→  }
   141→
   142→  const logout = () => {
   143→    setUser(null)
   144→    localStorage.removeItem('turborush_user')
   145→    setView('home')
   146→  }
   147→
   148→  if (!mounted) {
   149→    return (
   150→      <div className="min-h-screen flex items-center justify-center tr-bg">
   151→        <motion.div
   152→          animate={{ rotate: 360 }}
   153→          transition={{ repeat: Infinity, duration: 1, ease: "linear" }}
   154→          className="text-6xl"
   155→        >
   156→          🏎️
   157→        </motion.div>
   158→      </div>
   159→    )
   160→  }
   161→
   162→  if (!user) {
   163→    return <AuthScreen onLogin={login} />
   164→  }
   165→
   166→  return (
   167→    <AppShell user={user} view={view} setView={setView} onLogout={logout} />

   518→// ─── Dashboard View ───
   519→function DashboardView({ user, setView }: { user: User; setView: (v: View) => void }) {
   520→  const [stats, setStats] = useState<{ gamesPlayed: number; totalCorrect: number; totalCoins: number; bestStreak: number } | null>(null)
   521→  const [recentGames, setRecentGames] = useState<GameResult[]>([])
   522→  const [homeworks, setHomeworks] = useState<Homework[]>([])
   523→
   524→  useEffect(() => {
   525→    api.get(`/api/game-results?userId=${user.id}`).then((data: GameResult[]) => {
   526→      setRecentGames(data.slice(0, 5))
   527→      setStats({
   528→        gamesPlayed: data.length,
   529→        totalCorrect: data.reduce((s, g) => s + g.correct, 0),
   530→        totalCoins: data.reduce((s, g) => s + g.coinsEarned, 0),
   531→        bestStreak: Math.max(0, ...data.map(g => g.bestStreak)),
   532→      })
   533→    }).catch(() => {})
   534→    api.get('/api/homework').then(setHomeworks).catch(() => {})
   535→  }, [user.id])
   536→
   537→  const isTeacher = user.role === 'teacher'
   538→
   539→  return (
   540→    <div className="space-y-6">
   541→      {/* Welcome */}
   542→      <div className="tr-card p-6 lg:p-8">
   543→        <div className="flex flex-col lg:flex-row items-start lg:items-center justify-between gap-4">
   544→          <div>
   545→            <h1 className="tr-font-display text-2xl lg:text-3xl text-tr-navy">
   546→              {isTeacher ? '👩‍🏫' : '👋'} Hey, {user.name}!
   547→            </h1>
   548→            <p className="text-tr-navy/60 font-bold mt-1">
   549→              {isTeacher
   550→                ? 'Ready to create an engaging lesson?'
   551→                : 'Ready to race and learn?'}
   552→            </p>
   553→          </div>
   554→          <div className="flex gap-3">
   555→            {isTeacher ? (
   556→              <>
   557→                <button onClick={() => setView('live')} className="tr-btn text-sm">🎮 Host Game</button>
   558→                <button onClick={() => setView('create-set')} className="tr-btn tr-btn-green text-sm">📚 Create Set</button>
   559→              </>
   560→            ) : (
   561→              <>
   562→                <button onClick={() => setView('solo')} className="tr-btn text-sm">🏎️ Solo Play</button>

   803→// ─── Game Player (simplified for now — will port the full WebGL game) ───
   804→function GamePlayer({ questions, mode, playerName, userId, setId, onEnd }: {
   805→  questions: Question[]
   806→  mode: string
   807→  playerName: string
   808→  userId: string
   809→  setId: string
   810→  onEnd: () => void
   811→}) {
   812→  const [currentQ, setCurrentQ] = useState(0)
   813→  const [score, setScore] = useState(0)
   814→  const [correct, setCorrect] = useState(0)
   815→  const [wrong, setWrong] = useState(0)
   816→  const [streak, setStreak] = useState(0)
   817→  const [bestStreak, setBestStreak] = useState(0)
   818→  const [hearts, setHearts] = useState(3)
   819→  const [coins, setCoins] = useState(0)
   820→  const [showResult, setShowResult] = useState(false)
   821→  const [lastAnswer, setLastAnswer] = useState<'correct' | 'wrong' | null>(null)
   822→  const [gameOver, setGameOver] = useState(false)
   823→  const [timeLeft, setTimeLeft] = useState(15)
   824→  const [answered, setAnswered] = useState(false)
   825→
   826→  // Timer
   827→  useEffect(() => {
   828→    if (gameOver || answered) return
   829→    const timer = setInterval(() => {
   830→      setTimeLeft(prev => {
   831→        if (prev <= 1) {
   832→          handleAnswer(-1) // timeout
   833→          return 0
   834→        }
   835→        return prev - 1
   836→      })
   837→    }, 1000)
   838→    return () => clearInterval(timer)
   839→  }, [currentQ, gameOver, answered])
   840→
   841→  const handleAnswer = (idx: number) => {
   842→    if (answered) return
   843→    setAnswered(true)
   844→
   845→    const q = questions[currentQ]
   846→    const isCorrect = idx === q.correctIdx
   847→
   848→    if (isCorrect) {
   849→      const newStreak = streak + 1
   850→      const points = 50 + newStreak * 25
   851→      setScore(s => s + points)
   852→      setCorrect(c => c + 1)
   853→      setStreak(newStreak)
   854→      setBestStreak(bs => Math.max(bs, newStreak))
   855→      setCoins(c => c + 5)
   856→      setLastAnswer('correct')
   857→    } else {
   858→      setWrong(w => w + 1)
   859→      setStreak(0)
   860→      if (mode === 'heart') {
   861→        setHearts(h => {
   862→          const newH = h - 1
   863→          if (newH <= 0) setGameOver(true)
   864→          return newH
   865→        })
   866→      }
   867→      setLastAnswer('wrong')
   868→    }
   869→
   870→    setShowResult(true)
   871→    setTimeout(() => {
   872→      setShowResult(false)
   873→      setLastAnswer(null)
   874→      setAnswered(false)
   875→      setTimeLeft(15)
   876→      if (currentQ + 1 >= questions.length) {
   877→        setGameOver(true)
   878→      } else {
   879→        setCurrentQ(q => q + 1)
   880→      }
   881→    }, 1500)
   882→  }
   883→
   884→  // Save result on game over
   885→  useEffect(() => {
   886→    if (gameOver) {
   887→      api.post('/api/game-results', {
   888→        userId,
   889→        gameType: 'solo',
   890→        setId,
   891→        score,
   892→        correct,
   893→        wrong,
   894→        bestStreak,
   895→        coinsEarned: coins,
   896→      }).catch(() => {})
   897→    }
   898→  }, [gameOver])
   899→
   900→  if (gameOver) {
   901→    return (
   902→      <div className="tr-card p-8 text-center max-w-lg mx-auto">
   903→        <motion.div initial={{ scale: 0.5 }} animate={{ scale: 1 }} className="text-6xl mb-4">🏁</motion.div>
   904→        <h2 className="tr-font-display text-2xl text-tr-navy mb-4">Game Over!</h2>
   905→        <div className="grid grid-cols-2 gap-4 mb-6">
   906→          <div className="p-3 rounded-xl bg-tr-cream/50">
   907→            <p className="tr-font-display text-2xl text-tr-navy">{score.toLocaleString()}</p>
   908→            <p className="text-xs text-tr-navy/60 font-bold">Score</p>
   909→          </div>
   910→          <div className="p-3 rounded-xl bg-tr-grass/20">
   911→            <p className="tr-font-display text-2xl text-tr-navy">{correct}/{correct + wrong}</p>
   912→            <p className="text-xs text-tr-navy/60 font-bold">Correct</p>
   913→          </div>
   914→          <div className="p-3 rounded-xl bg-tr-yellow/20">
   915→            <p className="tr-font-display text-2xl text-tr-navy">{bestStreak}</p>
   916→            <p className="text-xs text-tr-navy/60 font-bold">Best Streak</p>
   917→          </div>
   918→          <div className="p-3 rounded-xl bg-tr-purple/20">
   919→            <p className="tr-font-display text-2xl text-tr-navy">🪙 {coins}</p>
   920→            <p className="text-xs text-tr-navy/60 font-bold">Coins Earned</p>
   921→          </div>
   922→        </div>
   923→        <div className="flex gap-3 justify-center">
   924→          <button onClick={onEnd} className="tr-btn text-sm">🏠 Dashboard</button>
   925→          <button onClick={() => {
   926→            setCurrentQ(0); setScore(0); setCorrect(0); setWrong(0); setStreak(0); setBestStreak(0); setHearts(3); setCoins(0); setGameOver(false); setAnswered(false); setTimeLeft(15)
   927→          }} className="tr-btn tr-btn-green text-sm">🔄 Play Again</button>
   928→        </div>
   929→      </div>
   930→    )
   931→  }
   932→
   933→  const q = questions[currentQ]
   934→  let choices: string[] = []
   935→  try { choices = JSON.parse(q.choices) } catch { choices = ['A', 'B', 'C', 'D'] }
   936→
   937→  const answerColors = ['bg-tr-red', 'bg-tr-sky', 'bg-tr-grass', 'bg-tr-yellow']
   938→
   939→  return (
   940→    <div className="max-w-2xl mx-auto space-y-4">
   941→      {/* HUD */}
   942→      <div className="tr-card p-4 flex items-center justify-between">
   943→        <div className="flex items-center gap-4">
   944→          <span className="tr-font-display text-tr-navy">Q{currentQ + 1}/{questions.length}</span>
   945→          <span className="tr-chip">🪙 {coins}</span>
   946→        </div>
   947→        <div className="flex items-center gap-4">
   948→          <span className="tr-font-display text-tr-navy">{score.toLocaleString()}</span>
   949→          {mode === 'heart' && (
   950→            <span className="text-lg">{Array(Math.max(0, hearts)).fill('❤️').join('')}</span>
   951→          )}
   952→          {streak > 1 && <span className="tr-chip">🔥 {streak}</span>}
   953→        </div>
   954→      </div>
   955→
   956→      {/* Timer */}
   957→      <div className="h-2 bg-tr-navy/10 rounded-full overflow-hidden">
   958→        <motion.div
   959→          animate={{ width: `${(timeLeft / 15) * 100}%` }}
   960→          transition={{ duration: 0.5 }}
   961→          className={`h-full rounded-full ${timeLeft > 5 ? 'bg-tr-grass' : 'bg-tr-red'}`}
   962→        />
   963→      </div>
   964→
   965→      {/* Question */}
   966→      <AnimatePresence mode="wait">
   967→        <motion.div
   968→          key={currentQ}
   969→          initial={{ opacity: 0, x: 50 }}
   970→          animate={{ opacity: 1, x: 0 }}
   971→          exit={{ opacity: 0, x: -50 }}
   972→          className="tr-card p-6"
   973→        >
   974→          <h2 className="tr-font-display text-xl text-tr-navy mb-6">{q.text}</h2>
   975→
   976→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   977→            {choices.map((choice, idx) => (
   978→              <motion.button
   979→                key={idx}
   980→                whileHover={!answered ? { scale: 1.02 } : {}}
   981→                whileTap={!answered ? { scale: 0.98 } : {}}
   982→                onClick={() => handleAnswer(idx)}
   983→                disabled={answered}
   984→                className={`p-4 rounded-xl border-3 font-bold text-white transition-all text-left ${
   985→                  answered && idx === q.correctIdx
   986→                    ? 'bg-tr-grass border-tr-navy ring-4 ring-tr-grass/30'
   987→                    : answered && lastAnswer === 'wrong' && idx !== q.correctIdx
   988→                    ? 'bg-tr-red/50 border-tr-navy opacity-50'
   989→                    : answerColors[idx] + ' border-tr-navy hover:shadow-lg'
   990→                }`}
   991→              >
   992→                <span className="text-sm opacity-80 block">{String.fromCharCode(65 + idx)}</span>
   993→                <span className="text-base">{choice}</span>
   994→              </motion.button>
   995→            ))}
   996→          </div>
   997→
   998→          {/* Result feedback */}
   999→          <AnimatePresence>
  1000→            {showResult && (
  1001→              <motion.div
  1002→                initial={{ opacity: 0, y: 10 }}
  1003→                animate={{ opacity: 1, y: 0 }}
  1004→                exit={{ opacity: 0 }}
  1005→                className={`mt-4 p-3 rounded-xl text-center font-bold ${
  1006→                  lastAnswer === 'correct' ? 'bg-tr-grass/20 text-tr-grass' : 'bg-tr-red/20 text-tr-red'
  1007→                }`}

  1009→                {lastAnswer === 'correct' ? '✅ Correct!' : `❌ ${q.explanation || 'The correct answer was ' + choices[q.correctIdx]}`}
  1010→              </motion.div>
  1011→            )}
  1012→          </AnimatePresence>
  1013→        </motion.div>
  1014→      </AnimatePresence>
