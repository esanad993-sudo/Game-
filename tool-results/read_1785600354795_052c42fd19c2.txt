   589→function pushBoxM(m,cx,cy,cz,sx,sy,sz,col){

   591→  for(let f=0;f<6;f++){
   592→    const F=FACES[f],b=m.vc/9;
   593→    for(let k=0;k<4;k++){
   594→      const a=CORN[k][0],c2=CORN[k][1];
   595→      V[m.vc++]=cx+F[0]*sx*.5+F[3]*a*sx*.5+F[6]*c2*sx*.5;
   596→      V[m.vc++]=cy+F[1]*sy*.5+F[4]*a*sy*.5+F[7]*c2*sy*.5;
   597→      V[m.vc++]=cz+F[2]*sz*.5+F[5]*a*sz*.5+F[8]*c2*sz*.5;
   598→      V[m.vc++]=F[0];V[m.vc++]=F[1];V[m.vc++]=F[2];
   599→      V[m.vc++]=col[0];V[m.vc++]=col[1];V[m.vc++]=col[2];
   600→    }
   601→    I[m.ic++]=b;I[m.ic++]=b+1;I[m.ic++]=b+2;I[m.ic++]=b;I[m.ic++]=b+2;I[m.ic++]=b+3;
