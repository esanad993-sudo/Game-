    94→// ─── API helpers ───
    95→const api = {
    96→  get: async (url: string) => {
    97→    const res = await fetch(url)
    98→    if (!res.ok) throw new Error(`GET ${url} failed`)
    99→    return res.json()
   100→  },
   101→  post: async (url: string, data: any) => {
   102→    const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   103→    if (!res.ok) { const err = await res.json().catch(() => ({})); throw new Error(err.error || `POST ${url} failed`) }
   104→    return res.json()
   105→  },
   106→  patch: async (url: string, data: any) => {
   107→    const res = await fetch(url, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   108→    if (!res.ok) throw new Error(`PATCH ${url} failed`)
   109→    return res.json()
   110→  },
   111→  del: async (url: string) => {
   112→    const res = await fetch(url, { method: 'DELETE' })
   113→    if (!res.ok) throw new Error(`DELETE ${url} failed`)
   114→    return res.json()
   115→  },
   116→}

   134→export default function Home() {
   135→  const [user, setUser] = useState<User | null>(null)
   136→  const [view, setView] = useState<View>('home')
   137→  const [mounted, setMounted] = useState(false)
   138→
   139→  useEffect(() => {
   140→    // Read from localStorage on mount - this is intentional initialization
   141→    const saved = localStorage.getItem('turborush_user')
   142→    if (saved) {
   143→      try {
   144→        const parsed = JSON.parse(saved)
   145→        // Use a microtask to avoid synchronous setState in effect
   146→        queueMicrotask(() => setUser(normalizeUser(parsed)))
   147→      } catch {}
   148→    }
   149→    queueMicrotask(() => setMounted(true))
   150→  }, [])
   151→
   152→  const login = (u: User) => {
   153→    const safe = normalizeUser(u)
   154→    setUser(safe)
   155→    localStorage.setItem('turborush_user', JSON.stringify(safe))
   156→  }
   157→
   158→  const logout = () => {
   159→    setUser(null)
   160→    localStorage.removeItem('turborush_user')
   161→    setView('home')
   162→  }
   163→
   164→  if (!mounted) {
   165→    return (
   166→      <div className="min-h-screen flex items-center justify-center tr-bg">
   167→        <motion.div
   168→          animate={{ rotate: 360 }}
   169→          transition={{ repeat: Infinity, duration: 1, ease: "linear" }}
   170→          className="text-6xl"
   171→        >
   172→          🏎️
   173→        </motion.div>
   174→      </div>
   175→    )
   176→  }
   177→
   178→  if (!user) {
   179→    return <AuthScreen onLogin={login} />
   180→  }
   181→
   182→  return (
   183→    <AppShell user={user} view={view} setView={setView} onLogout={logout} />
   184→  )

   532→// ─── Dashboard View ───
   533→function DashboardView({ user, setView }: { user: User; setView: (v: View) => void }) {
   534→  const [stats, setStats] = useState<{ gamesPlayed: number; totalCorrect: number; totalCoins: number; bestStreak: number } | null>(null)
   535→  const [recentGames, setRecentGames] = useState<GameResult[]>([])
   536→  const [homeworks, setHomeworks] = useState<Homework[]>([])
   537→
   538→  useEffect(() => {
   539→    api.get(`/api/game-results?userId=${user.id}`).then((data: GameResult[]) => {
   540→      setRecentGames(data.slice(0, 5))
   541→      setStats({
   542→        gamesPlayed: data.length,
   543→        totalCorrect: data.reduce((s, g) => s + g.correct, 0),
   544→        totalCoins: data.reduce((s, g) => s + g.coinsEarned, 0),
   545→        bestStreak: Math.max(0, ...data.map(g => g.bestStreak)),
   546→      })
   547→    }).catch(() => {})
   548→    api.get('/api/homework').then(setHomeworks).catch(() => {})
   549→  }, [user.id])
   550→
   551→  const isTeacher = user.role === 'teacher'
   552→
   553→  return (
   554→    <div className="space-y-6">
   555→      {/* Welcome */}
   556→      <div className="tr-card p-6 lg:p-8">
   557→        <div className="flex flex-col lg:flex-row items-start lg:items-center justify-between gap-4">
   558→          <div>
   559→            <h1 className="tr-font-display text-2xl lg:text-3xl text-tr-navy">
   560→              {isTeacher ? '👩‍🏫' : '👋'} Hey, {user.name}!
   561→            </h1>
   562→            <p className="text-tr-navy/60 font-bold mt-1">
   563→              {isTeacher
   564→                ? 'Ready to create an engaging lesson?'
   565→                : 'Ready to race and learn?'}
   566→            </p>
   567→          </div>
   568→          <div className="flex gap-3">

   817→// ─── Game Player (simplified for now — will port the full WebGL game) ───
   818→function GamePlayer({ questions, mode, playerName, userId, setId, onEnd }: {
   819→  questions: Question[]
   820→  mode: string
   821→  playerName: string
   822→  userId: string
   823→  setId: string
   824→  onEnd: () => void
   825→}) {
   826→  const [currentQ, setCurrentQ] = useState(0)
   827→  const [score, setScore] = useState(0)
   828→  const [correct, setCorrect] = useState(0)
   829→  const [wrong, setWrong] = useState(0)
   830→  const [streak, setStreak] = useState(0)
   831→  const [bestStreak, setBestStreak] = useState(0)
   832→  const [hearts, setHearts] = useState(3)
   833→  const [coins, setCoins] = useState(0)
   834→  const [showResult, setShowResult] = useState(false)
   835→  const [lastAnswer, setLastAnswer] = useState<'correct' | 'wrong' | null>(null)
   836→  const [gameOver, setGameOver] = useState(false)
   837→  const [timeLeft, setTimeLeft] = useState(15)
   838→  const [answered, setAnswered] = useState(false)
   839→
   840→  // Timer
   841→  useEffect(() => {
   842→    if (gameOver || answered) return
   843→    const timer = setInterval(() => {
   844→      setTimeLeft(prev => {
   845→        if (prev <= 1) {
   846→          handleAnswer(-1) // timeout
   847→          return 0
   848→        }
   849→        return prev - 1
   850→      })
   851→    }, 1000)
   852→    return () => clearInterval(timer)
   853→  }, [currentQ, gameOver, answered])
   854→
   855→  const handleAnswer = (idx: number) => {
   856→    if (answered) return
   857→    setAnswered(true)
   858→
   859→    const q = questions[currentQ]
   860→    const isCorrect = idx === q.correctIdx
   861→
   862→    if (isCorrect) {
   863→      const newStreak = streak + 1
   864→      const points = 50 + newStreak * 25
   865→      setScore(s => s + points)
   866→      setCorrect(c => c + 1)
   867→      setStreak(newStreak)
   868→      setBestStreak(bs => Math.max(bs, newStreak))
   869→      setCoins(c => c + 5)
   870→      setLastAnswer('correct')
   871→    } else {
   872→      setWrong(w => w + 1)
   873→      setStreak(0)
   874→      if (mode === 'heart') {
   875→        setHearts(h => {
   876→          const newH = h - 1
   877→          if (newH <= 0) setGameOver(true)
   878→          return newH
   879→        })
   880→      }
   881→      setLastAnswer('wrong')
   882→    }
   883→
   884→    setShowResult(true)
   885→    setTimeout(() => {
   886→      setShowResult(false)
   887→      setLastAnswer(null)
   888→      setAnswered(false)
   889→      setTimeLeft(15)
   890→      if (currentQ + 1 >= questions.length) {
   891→        setGameOver(true)
   892→      } else {
   893→        setCurrentQ(q => q + 1)
   894→      }
   895→    }, 1500)
   896→  }
   897→
   898→  // Save result on game over
   899→  useEffect(() => {
   900→    if (gameOver) {
   901→      api.post('/api/game-results', {
   902→        userId,
   903→        gameType: 'solo',
   904→        setId,
   905→        score,
   906→        correct,
   907→        wrong,
   908→        bestStreak,
   909→        coinsEarned: coins,
   910→      }).catch(() => {})
   911→    }
   912→  }, [gameOver])
   913→
   914→  if (gameOver) {
   915→    return (
   916→      <div className="tr-card p-8 text-center max-w-lg mx-auto">
   917→        <motion.div initial={{ scale: 0.5 }} animate={{ scale: 1 }} className="text-6xl mb-4">🏁</motion.div>
   918→        <h2 className="tr-font-display text-2xl text-tr-navy mb-4">Game Over!</h2>
   919→        <div className="grid grid-cols-2 gap-4 mb-6">
   920→          <div className="p-3 rounded-xl bg-tr-cream/50">
   921→            <p className="tr-font-display text-2xl text-tr-navy">{score.toLocaleString()}</p>
   922→            <p className="text-xs text-tr-navy/60 font-bold">Score</p>
   923→          </div>
   924→          <div className="p-3 rounded-xl bg-tr-grass/20">
   925→            <p className="tr-font-display text-2xl text-tr-navy">{correct}/{correct + wrong}</p>
   926→            <p className="text-xs text-tr-navy/60 font-bold">Correct</p>
   927→          </div>
   928→          <div className="p-3 rounded-xl bg-tr-yellow/20">
   929→            <p className="tr-font-display text-2xl text-tr-navy">{bestStreak}</p>
   930→            <p className="text-xs text-tr-navy/60 font-bold">Best Streak</p>
   931→          </div>
   932→          <div className="p-3 rounded-xl bg-tr-purple/20">
   933→            <p className="tr-font-display text-2xl text-tr-navy">🪙 {coins}</p>
   934→            <p className="text-xs text-tr-navy/60 font-bold">Coins Earned</p>
   935→          </div>
   936→        </div>
   937→        <div className="flex gap-3 justify-center">
   938→          <button onClick={onEnd} className="tr-btn text-sm">🏠 Dashboard</button>
   939→          <button onClick={() => {
   940→            setCurrentQ(0); setScore(0); setCorrect(0); setWrong(0); setStreak(0); setBestStreak(0); setHearts(3); setCoins(0); setGameOver(false); setAnswered(false); setTimeLeft(15)
   941→          }} className="tr-btn tr-btn-green text-sm">🔄 Play Again</button>
   942→        </div>
   943→      </div>
   944→    )
   945→  }
   946→
   947→  const q = questions[currentQ]
   948→  let choices: string[] = []
   949→  try { choices = JSON.parse(q.choices) } catch { choices = ['A', 'B', 'C', 'D'] }
   950→
   951→  const answerColors = ['bg-tr-red', 'bg-tr-sky', 'bg-tr-grass', 'bg-tr-yellow']
   952→
   953→  return (
   954→    <div className="max-w-2xl mx-auto space-y-4">
   955→      {/* HUD */}
   956→      <div className="tr-card p-4 flex items-center justify-between">
   957→        <div className="flex items-center gap-4">
   958→          <span className="tr-font-display text-tr-navy">Q{currentQ + 1}/{questions.length}</span>
   959→          <span className="tr-chip">🪙 {coins}</span>
   960→        </div>
   961→        <div className="flex items-center gap-4">
   962→          <span className="tr-font-display text-tr-navy">{score.toLocaleString()}</span>
   963→          {mode === 'heart' && (
   964→            <span className="text-lg">{Array(Math.max(0, hearts)).fill('❤️').join('')}</span>
   965→          )}
   966→          {streak > 1 && <span className="tr-chip">🔥 {streak}</span>}
   967→        </div>
   968→      </div>
   969→
   970→      {/* Timer */}
   971→      <div className="h-2 bg-tr-navy/10 rounded-full overflow-hidden">
   972→        <motion.div
   973→          animate={{ width: `${(timeLeft / 15) * 100}%` }}
   974→          transition={{ duration: 0.5 }}
   975→          className={`h-full rounded-full ${timeLeft > 5 ? 'bg-tr-grass' : 'bg-tr-red'}`}
   976→        />
   977→      </div>
   978→
   979→      {/* Question */}
   980→      <AnimatePresence mode="wait">
   981→        <motion.div
   982→          key={currentQ}
   983→          initial={{ opacity: 0, x: 50 }}
   984→          animate={{ opacity: 1, x: 0 }}
   985→          exit={{ opacity: 0, x: -50 }}
   986→          className="tr-card p-6"
   987→        >
   988→          <h2 className="tr-font-display text-xl text-tr-navy mb-6">{q.text}</h2>
   989→
   990→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   991→            {choices.map((choice, idx) => (
   992→              <motion.button
   993→                key={idx}
   994→                whileHover={!answered ? { scale: 1.02 } : {}}
   995→                whileTap={!answered ? { scale: 0.98 } : {}}
   996→                onClick={() => handleAnswer(idx)}
   997→                disabled={answered}
   998→                className={`p-4 rounded-xl border-3 font-bold text-white transition-all text-left ${
   999→                  answered && idx === q.correctIdx
  1000→                    ? 'bg-tr-grass border-tr-navy ring-4 ring-tr-grass/30'
  1001→                    : answered && lastAnswer === 'wrong' && idx !== q.correctIdx
  1002→                    ? 'bg-tr-red/50 border-tr-navy opacity-50'
  1003→                    : answerColors[idx] + ' border-tr-navy hover:shadow-lg'
  1004→                }`}
  1005→              >
  1006→                <span className="text-sm opacity-80 block">{String.fromCharCode(65 + idx)}</span>
  1007→                <span className="text-base">{choice}</span>
  1008→              </motion.button>
  1009→            ))}
  1010→          </div>
  1011→
  1012→          {/* Result feedback */}
  1013→          <AnimatePresence>
  1014→            {showResult && (
  1015→              <motion.div
  1016→                initial={{ opacity: 0, y: 10 }}
  1017→                animate={{ opacity: 1, y: 0 }}
  1018→                exit={{ opacity: 0 }}
  1019→                className={`mt-4 p-3 rounded-xl text-center font-bold ${
  1020→                  lastAnswer === 'correct' ? 'bg-tr-grass/20 text-tr-grass' : 'bg-tr-red/20 text-tr-red'
  1021→                }`}
  1022→              >
  1023→                {lastAnswer === 'correct' ? '✅ Correct!' : `❌ ${q.explanation || 'The correct answer was ' + choices[q.correctIdx]}`}
  1024→              </motion.div>
  1025→            )}
  1026→          </AnimatePresence>
  1027→        </motion.div>
  1028→      </AnimatePresence>
  1029→    </div>
