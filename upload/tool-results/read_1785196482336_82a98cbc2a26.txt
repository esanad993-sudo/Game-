  1173→  if(/^[1-4]$/.test(c)) return +c-1;
  1174→  if(/^[a-d]$/.test(c)) return c.charCodeAt(0)-97;

  1178→function parseQuestions(text){

  1180→  const lines=text.split(/\r?\n/).map(l=>l.trim()).filter(l=>l.length);
  1181→  const out=[], errs=[];
  1182→  for(let n=0;n<lines.length;n++){
  1183→    const cells=splitLine(lines[n],delim);
  1184→    if(n===0 && /question/i.test(cells[0]) && /correct|answer|ans|key/i.test(cells[cells.length-1])) continue;
  1185→    if(cells.length<3){ errs.push('Line '+(n+1)+': need question + answers + correct column'); continue; }
  1186→    const q=cells[0];
  1187→    const ans=cells.slice(1,cells.length-1).map(s=>s.trim()).filter(s=>s.length);
